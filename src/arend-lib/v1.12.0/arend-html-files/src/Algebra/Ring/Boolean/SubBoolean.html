\import Algebra.Ring.Boolean
\import Algebra.Ring.SubRing
\import Paths.Meta

\class SubBooleanPseudoRing \extends SubPseudoRing {
  \override S : BooleanRing
}

\instance IBooleanPseudoRing (S : SubBooleanPseudoRing {}) : BooleanRing
  | PseudoRing => IPseudoRing S
  | isBooleanRing => ext isBooleanRing
