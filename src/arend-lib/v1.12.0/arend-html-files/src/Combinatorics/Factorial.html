\import Algebra.Ordered
\import Arith.Nat
\import Function.Meta
\import Logic
\import Order.StrictOrder
\import Paths.Meta

\func fac (n : Nat) : Nat
  | 0 => 1
  | suc n => suc n Nat.* fac n

\lemma fac>0 {n : Nat} : 0 < fac n \elim n
  | 0 => NatOrder.zero<suc
  | suc n => rewrite NatSemiring.*-comm $ NatSemiring.<=_+-left zero<=_ fac>0

\lemma fac/=0 {n : Nat} : fac n /= 0
  => NatSemiring.>_/= fac>0
